
#include <stdio.h>
#include <string.h>

#define MAX 100

struct Book
{
    int id;
    char name[50];
    char author[50];
    int quantity;
};

struct Book books[MAX];
int count = 0;

// Function to add a book
void addBook()
{
    int i;

    if (count >= MAX)
    {
        printf("\nLibrary is full!\n");
        return;
    }

    printf("\nEnter Book ID: ");
    scanf("%d", &books[count].id);

    // Check duplicate ID
    for (i = 0; i < count; i++)
    {
        if (books[i].id == books[count].id)
        {
            printf("Book ID already exists!\n");
            return;
        }
    }

    printf("Enter Book Name: ");
    scanf(" %49[^\n]", books[count].name);

    printf("Enter Author Name: ");
    scanf(" %49[^\n]", books[count].author);

    printf("Enter Quantity: ");
    scanf("%d", &books[count].quantity);

    if (books[count].quantity < 0)
    {
        printf("Invalid quantity!\n");
        return;
    }

    count++;

    printf("\nBook added successfully!\n");
}

// Function to search a book
void searchBook()
{
    int id, i;

    printf("\nEnter Book ID to search: ");
    scanf("%d", &id);

    for (i = 0; i < count; i++)
    {
        if (books[i].id == id)
        {
            printf("\nBook Found!\n");
            printf("Book ID: %d\n", books[i].id);
            printf("Book Name: %s\n", books[i].name);
            printf("Author: %s\n", books[i].author);
            printf("Available Quantity: %d\n",
                   books[i].quantity);
            return;
        }
    }

    printf("\nBook not found!\n");
}

// Function to issue a book
void issueBook()
{
    int id, i;

    printf("\nEnter Book ID to issue: ");
    scanf("%d", &id);

    for (i = 0; i < count; i++)
    {
        if (books[i].id == id)
        {
            if (books[i].quantity > 0)
            {
                books[i].quantity--;
                printf("\nBook issued successfully!\n");
            }
            else
            {
                printf("\nBook is not available!\n");
            }
            return;
        }
    }

    printf("\nBook not found!\n");
}

// Function to return a book
void returnBook()
{
    int id, i;

    printf("\nEnter Book ID to return: ");
    scanf("%d", &id);

    for (i = 0; i < count; i++)
    {
        if (books[i].id == id)
        {
            books[i].quantity++;
            printf("\nBook returned successfully!\n");
            return;
        }
    }

    printf("\nBook not found!\n");
}

// Function to display available books
void displayBooks()
{
    int i, found = 0;

    printf("\n========== AVAILABLE BOOKS ==========\n");

    for (i = 0; i < count; i++)
    {
        if (books[i].quantity > 0)
        {
            printf("\nBook ID: %d\n", books[i].id);
            printf("Book Name: %s\n", books[i].name);
            printf("Author: %s\n", books[i].author);
            printf("Quantity: %d\n", books[i].quantity);
            found = 1;
        }
    }

    if (found == 0)
    {
        printf("\nNo books available!\n");
    }
}

// Main function
int main()
{
    int choice;

    do
    {
        printf("\n====================================\n");
        printf("  DIGITAL LIBRARY MANAGEMENT SYSTEM\n");
        printf("====================================\n");

        printf("1. Add Book\n");
        printf("2. Search Book by ID\n");
        printf("3. Issue Book\n");
        printf("4. Return Book\n");
        printf("5. Display Available Books\n");
        printf("6. Exit\n");

        printf("\nEnter your choice: ");
        scanf("%d", &choice);

        switch (choice)
        {
            case 1:
                addBook();
                break;

            case 2:
                searchBook();
                break;

            case 3:
                issueBook();
                break;

            case 4:
                returnBook();
                break;

            case 5:
                displayBooks();
                break;

            case 6:
                printf("\nThank you for using the Library System!\n");
                break;

            default:
                printf("\nInvalid choice! Try again.\n");
        }

    } while (choice != 6);

    return 0;
}
